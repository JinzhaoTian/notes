
# Redis Key

如何设计好一个key，


## 只读 Key

当 Node.js 服务器收到请求时，Express 会拦截该请求并将其转换为 `req` 对象。 因此，请求只不过是应用程序级别的 JavaScript 对象。

要生成与传入请求关联的 Redis 的 key，一个方法是将 `req` 对象序列化为 JSON 字符串，然后使用内置方法对其进行哈希处理。 但这个简单的解决方案并不是很有效，因为即使正文中的查询参数或字段的顺序不同时会得到不同的 key 值，比如，`/api/v1/users?offset=10&page=1` 和 `/api/v1/users?page=1&offeset=1`，但这是完全相同的 API 调用。 

这种 Redis 的 key 生成方法会导致 Redis 数据库存储空间过满，从而限制服务器的缓存能力。因此可以使用 `object-hash` 进行处理。
```js
const hash = require("object-hash");

const keyData = {
	query: req.query,
	body: req.body,
};

const cacheKey = `${req.path}@${hash.sha1(keyData)}`;
```


## 增删查改 Key

在增删查改中，对 key 的设计会略有不同。常用的设计是和业务相关，可以参考关系型数据库的表来设计，如：
1. 表名转换为 KEY 前缀
2. 第 2 段放置用于区分 KEY 的字段，对应 MySQL 中主键的列名
3. 第 3 段放置主键的值或者唯一键的值
4. 第 4 段写要存储的列名

```
set user@userid:10:username zhangsan
set user@userid:10:email    aa@126.com
set user@userid:10:age      20
```

## Promise 包装

为了使用方便，避免回调，可以使用 Promise 对 Redis 进行包装：

```js
const redis = require("redis");

const redisConfig = require("./config");
const logger = require("./logger");

class RedisClient {
    constructor() {
        this.redisClient = redis.createClient({
            host: redisConfig.redis_host,
            port: redisConfig.redis_port,
            password: redisConfig.redis_password,
            legacyMode: true,
        });

        this.redisClient.on("ready", () => {
            logger.info(`Redis Client : ready`);
        });

        this.redisClient.on("connect", () => {
            logger.info(`Redis Client : connect`);
        });

        this.redisClient.on("reconnecting", () => {
            logger.info(`Redis Client : reconnecting`);
        });

        this.redisClient.on("end", () => {
            logger.info(`Redis Client : close`);
        });

        this.redisClient.on("warning", () => {
            logger.info(`Redis Client : warning`);
        });

        this.redisClient.on("error", (err) => {
            logger.error(`Redis Client : Error, err: ${err.message}`);
        });

        if (this.redisClient.isOpen) {
            logger.info("Redis Client is connected.");
        } else {
            this.redisClient
                .connect()
                .then(() => {
                    logger.info("Redis Connection has been established successfully.");
                })
                .catch((err) => {
                    logger.error(`Unable to connect to the Redis, err: ${err.message}`);
                });
        }
    }

    async connect() {
        await this.redisClient.connect().catch((err) => {
            logger.error(`Unable to connect to the Redis by call, err: ${err.message}`);
        });
    }

    quit() {
        this.redisClient.quit();
    }

    async exists(key) {
        return new Promise((resolve, reject) => {
            this.redisClient.exists(key, (err, result) => {
                if (err) {
                    reject(false);
                }
                resolve(result);
            });
        });
    }

    async set(key, value, expire) {
        return new Promise((resolve, reject) => {
            this.redisClient.set(key, value, (err, result) => {
                if (err) {
                    reject(false);
                }
                if (!isNaN(expire)) {
                    this.redisClient.expire(key, expire);
                }
                resolve(result);
            });
        });
    }

    async get(key) {
        return new Promise((resolve, reject) => {
            this.redisClient.get(key, (err, result) => {
                if (err) {
                    reject(false);
                }
                resolve(result);
            });
        });
    }

    async remove(key) {
        return new Promise((resolve, reject) => {
            this.redisClient.del(key, (err, result) => {
                if (err) {
                    reject(false);
                }
                resolve(result);
            });
        });
    }

    async rPush(key, list, expires) {
        return new Promise((resolve, reject) => {
            this.redisClient.rPush(key, list, (err, result) => {
                if (err) {
                    reject(false);
                }
                if (!isNaN(expires)) {
                    this.redisClient.expire(key, expires);
                }
                resolve(result);
            });
        });
    }

    async lRange(key, startIndex = 0, stopIndex = -1) {
        return new Promise((resolve, reject) => {
            this.redisClient.lRange(key, startIndex, stopIndex, (err, result) => {
                if (err) {
                    reject(false);
                }
                resolve(result);
            });
        });
    }

    async lRem(key, n = 1, value) {
        return new Promise((resolve, reject) => {
            this.redisClient.lRem(key, n, value, (err, result) => {
                if (err) {
                    reject(false);
                }
                resolve(result);
            });
        });
    }
}

module.exports = new RedisClient();
```

## 业务整合

有两种方式可以整合到业务中，一种是直接写在 Service 里面，另一种是使用 Express 时，利用中间件进行补充。

### 写入 Service

```js
// entry.service.js

const RedisClient = require("../config/redis");
const EntryModel = require("../models/entry.model");

class EntryService {
    async getEntriesByPageWithCache(query, offset, limit) {
        const cacheKey = `entries-query:${JSON.stringify(query)}&&offset:${offset}&&limit:${limit}`;
        const expireTime = 60 + Math.floor(Math.random() * 10);

        const hasKey = await RedisClient.exists(cacheKey);
        if (hasKey) {
            const entriesStr = await RedisClient.get(cacheKey);
            return JSON.parse(entriesStr);
        } else {
            const entries = await EntryModel.findAll({ where: query, offset: offset, limit: limit });
            const entriesJson = entries.map((entry) => entry.toJSON());
            await RedisClient.set(cacheKey, JSON.stringify(entriesJson), expireTime);
            return entriesJson;
        }
    }

}
```

### 中间件

第二种我比较推荐，因为可以和业务逻辑分离：
```js
// cache.middleware.js

const logger = require("../config/logger");
const RedisClient = require("../config/redis");

const hash = require("object-hash");

async function cacheThroughReadMethod(req, res, next) {
    const keyData = {
        query: req.query,
        body: req.body,
    };

    const cacheKey = `${req.path}@${hash.sha1(keyData)}`;
    const expireTime = 60 + Math.floor(Math.random() * 10);
    const hasKey = await RedisClient.exists(cacheKey);
    if (hasKey) {
        const cacheResult = await RedisClient.get(cacheKey);
        logger.info(
            `Cache  Hited, call: "${req.method} ${req.url}@${JSON.stringify(keyData)}", cacheKey: "${cacheKey}"`
        );
        res.json(JSON.parse(cacheResult));
    } else {
        logger.info(
            `Cache  Missed, call: "${req.method} ${req.url}@${JSON.stringify(keyData)}", cacheKey: "${cacheKey}"`
        );
        const oldJSON = res.json;
        res.json = async (data) => {
            res.json = oldJSON;
            if (res.statusCode.toString().startsWith("2")) {
                await RedisClient.set(cacheKey, JSON.stringify(data), expireTime);
                logger.info(
                    `Cache  Writed, call: "${req.method} ${req.url}@${JSON.stringify(
                        keyData
                    )}", cacheKey: "${cacheKey}"`
                );
            }
            return res.json(data);
        };
        next();
    }
}

async function cacheThroughWriteMethod(req, res, next) {
    const keyData = {
        query: req.query,
        body: req.body,
    };

    const cacheKey = `${req.path}@${hash.sha1(keyData)}`;
    const oldJSON = res.json;
    res.json = async (data) => {
        res.json = oldJSON;
        if (res.statusCode.toString().startsWith("2")) {
            const hasKey = await RedisClient.exists(cacheKey);
            if (hasKey) {
                await RedisClient.remove(cacheKey);
                logger.info(
                    `Cache  Deleted, call: "${req.method} ${req.url}@${JSON.stringify(
                        keyData
                    )}", cacheKey: "${cacheKey}"`
                );
            }
        }
        return res.json(data);
    };
    next();
}

function redisCache(req, res, next) {
    if (req.method === "GET") {
        cacheThroughReadMethod(req, res, next);
    } else if (req.method === "POST" || req.method === "PUT" || req.method === "DELETE") {
        cacheThroughWriteMethod(req, res, next);
    } else {
        next();
    }
}

module.exports = redisCache;
```


这样在主函数中，就可以直接使用：
```js
const redisCache = require("./middlewares/cache.middleware");

// 使用 redis cache
app.use(redisCache);

app.use("/api", [
    //require("./controllers/user.controller"),
    require("./controllers/template.controller"),
    require("./controllers/entry.controller"),
    require("./controllers/project.controller"),
    require("./controllers/parameter.controller"),
]);
```
