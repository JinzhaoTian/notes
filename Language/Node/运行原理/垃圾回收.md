TypeScript 和 JavaScript 的[垃圾回收](../../JavaScript/JavaScript%20垃圾回收.md)机制取决于它们运行的引擎（如 V8），而 Node.js 使用 V8 引擎，所以它们的基本原理相同，但 Node.js 可能会根据服务器端的特点进行一些调整和优化。


## 现代 V8 引擎的分代垃圾回收

V8 将堆内存分为两个主要区域：
1. **新生代（Young Generation）**
    - 使用 Scavenge 算法
    - 分为 From-space 和 To-space
    - 频繁进行垃圾回收（复制存活对象）
2. **老生代（Old Generation）**
    - 使用标记清除和标记压缩算法
    - 存储长期存活的对象
    - 垃圾回收频率较低但耗时较长

## Node.js 的垃圾回收机制

> [!tip]- Node.js 特有的考虑因素
> ```javascript
> // 服务器端需要考虑的内存管理
> const server = require('http').createServer();
> 
> // 1. 长连接可能导致内存泄漏
> let connections = new Set();
> 
> server.on('connection', (socket) => {
>     connections.add(socket);
>     socket.on('close', () => {
>         connections.delete(socket); // 必须手动删除
>     });
> });
>
> // 2. 大对象缓存需要谨慎处理
> const cache = new Map();
> 
> function getData(key) {
>     if (cache.has(key)) {
>         return cache.get(key);
>     }
>    const data = expensiveOperation();
>    cache.set(key, data);
>    
>    // 防止缓存无限增长
>     if (cache.size > 1000) {
>         const firstKey = cache.keys().next().value;
>         cache.delete(firstKey);
>     }
>     return data;
> }
> ```

Node.js 使用 V8 引擎，因此继承其垃圾回收机制，但有特定优化：
1. **垃圾回收配置**：
```bash
# 启动时设置内存限制
node --max-old-space-size=4096 app.js  # 设置老生代内存为4GB
node --max-semi-space-size=64 app.js   # 设置新生代半空间为64MB

# 启用垃圾回收日志
node --trace-gc app.js
node --trace-gc --trace-gc-verbose app.js

# 性能分析
node --prof app.js
node --heapsnapshot-signal=SIGUSR2 app.js
```

### 性能优化

1. **减少垃圾回收压力**
```javascript
// 1. 对象池模式
class ObjectPool {
    constructor(createFn) {
        this.createFn = createFn;
        this.pool = [];
    }
    
    acquire() {
        return this.pool.length > 0 ? this.pool.pop() : this.createFn();
    }
    
    release(obj) {
        // 重置对象状态而非删除
        this.pool.push(obj);
    }
}

// 2. 避免频繁创建临时对象
function inefficient(str) {
    return str.split('').reverse().join(''); // 创建多个临时数组
}

function efficient(str) {
    let result = '';
    for (let i = str.length - 1; i >= 0; i--) {
        result += str[i]; // 减少临时对象
    }
    return result;
}

// 3. 使用 Buffer 而非字符串处理二进制数据
const buffer = Buffer.alloc(1024); // 固定内存，不触发频繁GC
```

2. **监控内存使用**
```javascript
// Node.js 内存监控
const v8 = require('v8');

setInterval(() => {
    const memoryUsage = process.memoryUsage();
    console.log({
        rss: Math.round(memoryUsage.rss / 1024 / 1024) + 'MB',
        heapTotal: Math.round(memoryUsage.heapTotal / 1024 / 1024) + 'MB',
        heapUsed: Math.round(memoryUsage.heapUsed / 1024 / 1024) + 'MB',
        external: Math.round(memoryUsage.external / 1024 / 1024) + 'MB',
    });
    
    // 堆统计
    const heapStats = v8.getHeapStatistics();
    console.log('堆统计:', heapStats);
}, 5000);

// 创建堆快照用于分析
const fs = require('fs');
const { writeHeapSnapshot } = require('v8');

function takeHeapSnapshot() {
    const snapshotPath = `./heapdump-${Date.now()}.heapsnapshot`;
    writeHeapSnapshot(snapshotPath);
    console.log(`堆快照已保存: ${snapshotPath}`);
}
```
