### C++

```c++
void mergeOp(vector<int>& arr, int low, int mid, int high) {
    int i = low, j = mid + 1;
    vector<int> ans;
    while(i <= mid && j <= high) {
        if (arr[i] < arr[j])
            ans.push_back(arr[i++]);
        else
            ans.push_back(arr[j++]);
    }
    while (i <= mid)
        ans.push_back(arr[i++]);
    while (j <= high)
        ans.push_back(arr[j++]);
    int k = 0;
    for(i = low, k = 0; i <= high; ++i, ++k)
        arr[i] = ans[k];
}

void mergeSort(vector<int>& arr, int low, int high) {
    if (low < high) {
        int mid = low + (high - low) / 2;
        mergeSort(arr, low, mid);
        mergeSort(arr, mid + 1, high);
        mergeOp(arr, low, mid, high);
    }
}
```
