
### KMP

核心思想就是模式匹配时，指向主串s的 i 始终不后退，后退的是指向pattern的 j ，同时并不后退到没有用的位置。

```python
def KMP(s, pattern):
    nex = getNext(pattern)
    i, j = 0, 0
    while i < len(s) and j < len(pattern):
        if j == -1 or s[i] == pattern[j]: # j==-1是由于j=next[j]产生
            i += 1
            j += 1
        else:
            j = nex[j]

    if j == len(pattern): # j走到了末尾，说明匹配到了
        return i - j
    else:
        return -1

def getNext(pattern):
    nex = [0] * (len(pattern) + 1)
    nex[0] = -1
    i = 0
    j = -1
    while i < len(pattern):
        if j == -1 or pattern[i] == pattern[j]:
            i += 1
            j += 1
            nex[i] = j     # 这是最大的不同：记录next[i]
        else:
            j = nex[j]
    return nex
```

C++版本：

```python
vector<int> getNext(string pattern) {
    vector<int> nex(pattern.size() + 1, 0);
    nex[0] = -1;
    int i = 0, j = -1;
    while (i < pattern.size()) {
        if (j == -1 || pattern[j] == pattern[i]) {
            i ++;
            j ++;
            nex[i] = j;
        } else {
            j = nex[j];
        }
    }
    return nex;
}

int kmp(string s, string pattern) {
    vector<int> nex = getNext(pattern);
    int slen = (int)s.size(), plen = (int)pattern.size();
    int i = 0, j = 0;
    while (i < slen && j < plen) {
        if (j == -1 || s[i] == pattern[j]) {
            i++;
            j++;
        } else {
            j = nex[j];
        }
    }
    if (j == plen)
        return i - j;
    else
        return -1;
}
```
