void adjust(vector<int>& arr, int n, int idx) {
    int left = 2 * idx + 1, right = 2 * idx + 2;
    int m = idx;
    if (left < n && arr[left] > arr[m])
        m = left;
    if (right < n && arr[right] > arr[m])
        m = right;
    
    if (m != idx) {
        swap(arr[m], arr[idx]);
        adjust(arr, n, m);                  // 从上到下调整
    }
}

void heapSort(vector<int>& arr, int n) {
    for (int i = n / 2 - 1; i >= 0; --i) {
        adjust(arr, n, i);
    }
    
    for (int i = n - 1; i >= 1; i--) {
        swap(arr[0], arr[i]);               // 把最大的放在末尾
        adjust(arr, i, 0);                  // 重新调整前几个
    }
}
